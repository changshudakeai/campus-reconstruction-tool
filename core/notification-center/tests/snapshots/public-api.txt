## Enums #[non_exhaustive]

pub enum NotificationLevel { Error, Warn, Info }
pub enum NotificationActionOutcome { Succeeded(Notification), Failed(Notification) }

## Structures

pub struct Notification { pub id: Uuid, pub level: NotificationLevel, pub title: String, pub body: String, pub source_tag: String, pub created_at: DateTime<Utc> }
pub struct OpaqueNotificationAction { .. }
pub struct NotificationRecord { .. }
pub struct InMemoryStorage { .. }
pub struct DummyPresenter
pub struct PresenterRegistry { .. }
pub struct NotificationCenter { .. }
pub struct CleanupScheduler { .. }

## Traits

pub trait Storage: Send + Sync {
    fn append(&self, notification: Notification)
    fn all_notifications(&self) -> Vec<Notification>
    fn snapshot(&self) -> Vec<Notification>
    fn remove_by_id(&self, id: &Uuid)
}

pub trait Presenter: Send + Sync {
    fn show_error_dialog(&self, notification: &Notification)   // 必须模态阻塞
    fn show_toast(&self, notification: &Notification)          // 不得阻塞
    fn update_unread_count(&self, count: usize)
}

## Implementations

impl NotificationLevel {
    pub fn as_str(&self) -> &'static str
    pub fn is_error(&self) -> bool
    pub fn is_warn(&self) -> bool
    pub fn is_info(&self) -> bool
}

impl OpaqueNotificationAction {
    pub fn new<Outcome>(executor: impl Fn() -> Outcome + Send + Sync + 'static) -> Self where Outcome: Into<NotificationActionOutcome>
    pub fn invoke(&self) -> NotificationActionOutcome
}

impl NotificationActionOutcome {
    pub fn succeeded(notification: Notification) -> Self
    pub fn failed(notification: Notification) -> Self
    pub fn is_failed(&self) -> bool
    pub fn into_notification(self) -> Notification
}

impl NotificationRecord {
    pub fn new(notification: Notification) -> Self
    pub fn with_diagnostic_action(self, action: OpaqueNotificationAction) -> Self
    pub fn notification(&self) -> &Notification
    pub fn has_diagnostic_action(&self) -> bool
    pub fn diagnostic_action(&self) -> Option<&OpaqueNotificationAction>
}
impl Notification {
    pub fn error(source_tag: impl Into<String>, title: impl Into<String>, body: impl Into<String>) -> Self
    pub fn warn(source_tag: impl Into<String>, title: impl Into<String>, body: impl Into<String>) -> Self
    pub fn info(source_tag: impl Into<String>, title: impl Into<String>, body: impl Into<String>) -> Self
    pub fn is_expired(&self) -> bool
    pub fn clipboard_text(&self) -> String
}

impl InMemoryStorage {
    pub fn new() -> Self
}
impl Storage for InMemoryStorage

impl Presenter for DummyPresenter

impl PresenterRegistry {
    pub fn new() -> Self
    pub fn set_presenter(&self, presenter: Arc<dyn Presenter>)
    pub fn clear_presenter(&self)
    pub fn show_error(&self, notification: &Notification)
    pub fn show_toast(&self, notification: &Notification)
    pub fn update_unread_count(&self, count: usize)
    pub fn storage(&self) -> &InMemoryStorage
}

impl NotificationCenter {
    pub fn new(registry: PresenterRegistry) -> Self
    pub fn init(registry: PresenterRegistry) -> Arc<Self>
    pub fn global() -> Option<Arc<Self>>
    pub fn publish(&self, notification: Notification)
    pub fn publish_with_action(&self, notification: Notification, action: OpaqueNotificationAction)
    pub fn publish_action_outcome(&self, outcome: NotificationActionOutcome, is_current: bool)
    pub fn board_records(&self) -> Vec<NotificationRecord>
    pub fn diagnostic_action(&self, notification_id: &str) -> Option<OpaqueNotificationAction>
    pub fn board_snapshot(&self) -> Vec<Notification>
    pub fn mark_board_opened(&self)
    pub fn unread_count(&self) -> usize
    pub fn registry(&self) -> &PresenterRegistry
    pub fn cleanup(&self) -> &CleanupScheduler
}

impl CleanupScheduler {
    pub fn new(storage: Arc<dyn Storage>) -> Self
    pub fn start(&self)
    pub fn start_with_interval(&self, interval: Duration)
    pub fn stop(&self)
    pub fn is_running(&self) -> bool
    pub fn run_once(&self)
}
impl Drop for CleanupScheduler
