## T24: 边界编辑地图页与 IPC 协议

pub struct BoundaryEditPageConfig {                          (Debug + Clone + PartialEq)
    pub api_key: String,
    pub security_key: String,
    pub anchor_lon: f64,
    pub anchor_lat: f64,
    pub height_px: u32,
}

pub fn BoundaryEditPageConfig::with_anchor(self, lon: f64, lat: f64) -> Self
pub fn BoundaryEditPageConfig::effective_height_px(&self) -> u32  (钳制 >= 300)

pub enum IpcMessage {                                        (Debug + Clone + PartialEq)
    Coordinate { longitude: f64, latitude: f64 },            (T23 pick point / 纯文本经纬度)
    Error { message: String },
    OsmElements { elements: Vec<OsmElement> },               (T24: Overpass 原始要素回传)
    BoundaryUpdate { coords: Vec<[f64; 2]> },                (T24: 编辑中坐标 GCJ-02)
    ManualPoint { lon: f64, lat: f64 },                      (T24: 人工圈画落点)
    ManualCancel,                                            (T24: 撤销落点)
    ManualClear,                                             (T24: 清空重画)
    ConfirmBoundary { coords: Vec<[f64; 2]> },               (T24: 确认最终边界 GCJ-02)
    BoundaryGeometryUpdate { r#type: String, coordinates: serde_json::Value }, (GeoJSON Polygon/MultiPolygon 编辑中)
    ConfirmBoundaryGeometry { r#type: String, coordinates: serde_json::Value }, (GeoJSON Polygon/MultiPolygon 最终确认)
}

pub struct OsmElement {                                      (Serialize + Deserialize + PartialEq)
    pub r#type: String,
    pub id: i64,
    pub geometry: Option<Vec<[f64; 2]>>,
    pub members: Vec<OsmMember>,
    pub tags: HashMap<String, String>,
}
pub struct OsmMember {                                       (Serialize + Deserialize + PartialEq)
    pub r#type: String,
    pub reference: i64,
    pub role: String,
}

}

    pub element: OsmElement,

    pub name_match_score: f64,
    pub distance_to_anchor_m: f64,

}

    pub fn new(element: OsmElement) -> Self
}

}
