## Traits & Implementations

impl Database {
    pub fn open(path: impl AsRef<Path>) -> Result<Self>
    pub fn open_in_memory() -> Result<Self>
    pub fn schema_version(&self) -> Result<Option<u32>>
    pub fn delete_plan_to_trash(&mut self, campus_id: &str, plan_id: &str) -> Result<TrashItem>
    pub fn purge_plan_permanently(&mut self, trash_id: &str) -> Result<()>
    pub fn purge_expired_plans(&mut self) -> Result<usize>
}

impl RawObservation {
    pub fn new(plan_id, entity_type, entity_id, source_data, data_source_tag) -> Self
    pub fn compute_digest(source_data: &serde_json::Value) -> String
}

impl ReviewDecision {
    pub fn new(plan_id, entity_type, entity_id, review_state) -> Self
}

impl TrashItem {
    pub fn new_plan(campus_id, plan_id, deleted_by: Option<String>) -> Self
    pub fn is_restorable(&self, now: DateTime<Utc>) -> bool
}

pub trait RawObservationsApi {                   (impl for Database)
    fn write_raw_observations(&mut self, observations: &[RawObservation]) -> Result<usize>
    fn list_raw_observations(&self, plan_id: &str) -> Result<Vec<RawObservation>>
    fn list_raw_observations_by_category(&self, plan_id: &str, category: CandidateCategory) -> Result<Vec<RawObservation>>
    fn find_raw_observation(&self, plan_id: &str, category: CandidateCategory, entity_id: &str) -> Result<Option<RawObservation>>
}

pub trait ReviewDecisionsApi {                   (impl for Database)
    fn batch_update_review_decisions(&mut self, decisions: &[ReviewDecision]) -> Result<()>
    fn list_review_decisions(&self, plan_id: &str) -> Result<Vec<ReviewDecision>>
    fn count_review_states(&self, plan_id: &str) -> Result<(usize, usize, usize)>
}

pub trait CampusCrudApi {                        (impl for Connection/Database)
    fn list_campuses(&self) -> Result<Vec<CampusEntity>>
    fn create_campus(&mut self, name: &str) -> Result<CampusEntity>
    fn create_campus_with_anchor(&mut self, name: &str, poi_id: &str, address: &str, anchor_lng: f64, anchor_lat: f64) -> Result<CampusEntity>
    fn find_campus_by_id(&self, campus_id: &str) -> Result<Option<CampusEntity>>
    fn update_campus_anchor(&self, campus_id: &str, anchor_lng: f64, anchor_lat: f64) -> Result<()>
}

pub trait AppSettingsApi {                       (impl for Connection/Database)
    fn get_setting(&self, key: AppSettingKey) -> Result<Option<String>>
    fn set_setting(&mut self, key: AppSettingKey, value: &str) -> Result<()>
}

pub trait PlanCrudApi {                          (impl for Connection/Database)
    fn list_plans(&self, campus_id: &str) -> Result<Vec<PlanEntity>>
    fn create_plan(&mut self, campus_id: &str, name: &str) -> Result<PlanEntity>
    fn rename_plan(&mut self, plan_id: &str, new_name: &str) -> Result<()>
    fn find_plan_by_id(&self, plan_id: &str) -> Result<Option<PlanEntity>>
    fn touch_plan(&mut self, plan_id: &str) -> Result<()>
}

pub trait TrashApi {                             (impl for Database)
    fn insert_to_trash(&mut self, item: &TrashItem) -> Result<()>
    fn list_restorable_trash(&self, campus_id: &str) -> Result<Vec<TrashItem>>
    fn restore_from_trash(&mut self, id: &str) -> Result<TrashItem>
    fn permanently_delete(&mut self, id: &str) -> Result<()>
    fn purge_expired(&mut self) -> Result<usize>
    fn purge_all_in_campus_trash(&mut self, campus_id: &str) -> Result<usize>   (S1-04：清空当前校区回收站)
}

## 铁律备注

- 数据粮仓：本 crate 不存在、也永不提供删除 raw_observations 的 API
- 批量写回一律单事务原子提交（失败整批回滚）
- 回收站到期清理只提供框架（purge_expired），调用时机由功能层决定
- 清空回收站（purge_all_in_campus_trash）只清理当前校区的仍可恢复条目，不影响其他校区（ADR-0018 §三）
