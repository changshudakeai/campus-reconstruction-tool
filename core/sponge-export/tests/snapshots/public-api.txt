## Structures

pub struct VoxelModel {
    pub width: usize,
    pub height: usize,
    pub length: usize,
    pub palette: Vec<String>,
    pub blocks: Vec<u16>,
}

pub struct SchematicProfile { pub minecraft_version: &'static str, pub data_version: i32 }

pub struct SchematicInspection {
    pub sponge_version: i32,
    pub data_version: i32,
    pub dimensions: [usize; 3],
    pub offset: [i32; 3],
    pub palette_entries: usize,
    pub total_voxels: usize,
    pub non_air_voxels: usize,
    pub voxels: Vec<u32>,
    pub palette: HashMap<String, i32>,
    pub content_sha256: String,
}

## Functions

pub fn write_schematic(path: &Path, name: &str, model: &VoxelModel) -> Result<(), anyhow::Error>

pub fn encode_schematic(name: &str, model: &VoxelModel, profile: SchematicProfile) -> Result<Vec<u8>, anyhow::Error>
pub fn inspect_schematic(path: &Path) -> Result<SchematicInspection, anyhow::Error>
pub fn verify_worldedit_import_contract(path: &Path) -> Result<SchematicInspection, anyhow::Error>
pub fn verify_schematic_profile(path: &Path, profile: SchematicProfile) -> Result<SchematicInspection, anyhow::Error>
