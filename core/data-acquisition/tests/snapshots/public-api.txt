## Enums

pub enum DiffKind { Added, Updated, Unchanged }
pub enum SourceGeometry { Point((f64, f64)), LineString(Vec<(f64, f64)>), Polygon(Vec<(f64, f64)>) }
pub enum AcquisitionError {                                #[non_exhaustive]
    EmptyBoundary,
    SourceUnreachable { source_tag: String, message: String },

## Traits & Implementations

pub trait DataSource {
    fn source_tag(&self) -> &str
    fn fetch_raw_entities(&self, boundary: &Boundary) -> Result<Vec<RawEntity>>
}

impl RawEntity {
    pub fn new(entity_id: impl Into<String>, name: impl Into<String>, tags: TagMap, source_payload: serde_json::Value) -> Self
    pub fn with_geometry(...) -> Self
    pub fn to_source_data(&self) -> serde_json::Value
}

impl GaodeDataSource {
    pub const SOURCE_TAG: &'static str
    pub fn new(transport: BridgeTransport) -> Self
}
impl DataSource for GaodeDataSource
impl OverpassDataSource { pub const SOURCE_TAG: &'static str; pub fn new(transport: OverpassTransport) -> Self }
impl DataSource for OverpassDataSource

impl AcquisitionPipeline {
    pub fn new() -> Result<Self>
    pub fn with_engine(engine: ClassifyEngine) -> Self
    pub fn engine(&self) -> &ClassifyEngine
    pub fn collect(&self, db: &mut Database, plan_id: &PlanId, boundary: &Boundary, source: &dyn DataSource) -> Result<CollectionReport>
    pub fn acquire_batch(&self, db: &mut Database, plan_id: &PlanId, boundary: &Boundary, source: &dyn DataSource) -> Result<AcquisitionBatch>
}

impl DiffKind {
    pub fn text_key(&self) -> &'static str
}

impl RefreshDiff {
    pub fn new(entries: Vec<DiffEntry>) -> Self
    pub fn entries(&self) -> &[DiffEntry]
    pub fn added_count(&self) -> usize
    pub fn updated_count(&self) -> usize
    pub fn unchanged_count(&self) -> usize
    pub fn has_changes(&self) -> bool
    pub fn summary_key(&self) -> &'static str
}

impl CollectionProgressView {
    pub fn fetching() -> Self
    pub fn completed(report: &CollectionReport) -> Self
}
