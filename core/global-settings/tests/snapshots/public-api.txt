# global-settings Public API Snapshot
#
# 本文件记录 F1 应用全局设置的公开 API 结构
# 任何增删都会在此文件中体现，PR diff 可见
#
# 最后更新: 2026-07-31 (S1-04 最近使用记录)

## Constants

pub const SUPPORTED_LANGUAGES: &[&str] = &["zh-CN"]           (ADR-0004：当前仅中文)
pub const SUPPORTED_MINECRAFT_VERSIONS: &[&str] = &["26.1.2"] (ADR-0004：当前仅 26.1.2)
pub const DEFAULT_LANGUAGE: &str = "zh-CN"
pub const DEFAULT_MINECRAFT_VERSION: &str = "26.1.2"
pub const VERSION_NOTICE_TEXT: &str                           (首次设置页知情告知文字)

## Structures

pub struct GlobalSettings {
    pub language: String,
    pub minecraft_version: String,
}
impl Default for GlobalSettings                               (zh-CN / 26.1.2)

pub struct FirstRunSetup {
    pub language: String,
    pub minecraft_version: String,
    pub acknowledged: bool,
}
impl Default for FirstRunSetup                                (默认未勾选知情告知)

pub struct LandingCampus {
    pub id: CampusId,                                         (B1 共享类型，T02 复用)
    pub name: String,
}
pub struct RecentCampus {
    pub id: CampusId,
    pub name: String,
    pub address: String,                                      (S1-04：最近记录卡片地址，ADR-0006)
}

pub enum StartupDestination { FirstRunSetup, CampusSelect, LastUsedCampus { name: String } }
pub struct StartupSnapshot {
    pub settings: GlobalSettings,
    pub destination: StartupDestination,
}
pub struct CompleteStartupResult<LandingContent> {
    pub snapshot: StartupSnapshot,
    pub landing_content: Option<LandingContent>,
}
pub enum StartupResultError<LandingError> {
    Startup(Error),
    Landing(LandingError),
}

## Traits

pub trait StartupLandingContentProvider {
    type Content
    type Error
    fn landing_content(&self) -> std::result::Result<Self::Content, Self::Error>
}
pub struct SettingsSnapshot {
    pub settings: GlobalSettings,
    pub gaode_api_key: Option<String>,
    pub gaode_security_key: Option<String>,
    pub default_export_location: String,
}
pub struct SettingsManager { .. }                             (内部持有 Database，不外泄)

## SettingsManager Methods

pub fn new(db: Database) -> Self
pub fn is_first_run(&self) -> Result<bool>
pub fn settings(&self) -> Result<GlobalSettings>
pub fn startup_result<Provider>(&self, provider: &Provider) -> std::result::Result<CompleteStartupResult<Provider::Content>, StartupResultError<Provider::Error>> where Provider: StartupLandingContentProvider
pub fn settings_snapshot(&self) -> Result<SettingsSnapshot>
pub fn default_export_location(&self) -> Result<String>
pub fn set_default_export_location(&mut self, path: &str) -> Result<()>
pub fn set_language(&mut self, language: &str) -> Result<()>
pub fn set_minecraft_version(&mut self, version: &str) -> Result<()>
pub fn complete_first_run(&mut self, setup: &FirstRunSetup) -> Result<()>
pub fn landing_campus(&self) -> Result<Option<LandingCampus>>  (ADR-0006：被删/未设置 → None)
pub fn remember_campus(&mut self, campus_id: &CampusId) -> Result<()>  (S1-04：同步维护最近列表，最近进入排最前)
pub fn recent_campuses(&self) -> Result<Vec<RecentCampus>>     (S1-04：最近使用记录，ADR-0006)
pub fn remove_recent_campus(&mut self, campus_id: &CampusId) -> Result<()>  (S1-04：右侧小叉移除快捷记录)
pub fn select_campus_with_anchor(&mut self, name: &str, poi_id: &str, address: &str, anchor_lng: f64, anchor_lat: f64) -> Result<CampusId>  (S1-04：新增地址参数)
pub fn update_campus_anchor(&self, campus_id: &CampusId, anchor_lng: f64, anchor_lat: f64) -> Result<()>
pub fn gaode_api_key(&self) -> Result<Option<String>>
pub fn gaode_security_key(&self) -> Result<Option<String>>
pub fn set_gaode_api_key(&mut self, key: &str) -> Result<()>
pub fn set_gaode_security_key(&mut self, key: &str) -> Result<()>
pub fn clear_gaode_keys(&mut self) -> Result<()>
pub fn test_gaode_connection(&self, api_key: &str, security_key: &str) -> Result<()>

## Errors

pub type Result<T> = std::result::Result<T, Error>
pub enum Error {                                              (#[non_exhaustive])
    Storage(data_persistence::Error),
    UnsupportedLanguage(String),
    UnsupportedMinecraftVersion(String),
    NoticeNotAcknowledged,
    MissingGaodeKeys(String),                                (首启向导必填 Key 缺失)
    InvalidGaodeApiKey,
    InvalidGaodeSecurityKey,
    GaodeConnectionFailed(String),
    InvalidExportLocation,
    InvalidCampusId(String),
    InvalidRecentCampuses(String),                            (S1-04：最近记录 JSON 损坏)
}
