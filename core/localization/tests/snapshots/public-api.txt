## Enums

pub enum Language { ZhCn }
    impl Default for Language  # 默认 ZhCn

## Structures

pub struct Localization  # 字段私有：language + resources 映射表

## Implementations

impl Localization {
    pub fn new(language: Language) -> Result<Self, String>  # 加载顺序：可执行文件旁 resources/ → cwd resources/ → 内嵌副本兑底
    pub fn language(&self) -> Language
    pub fn t(&self, key: &str) -> String  # 简单文本键翻译；缺失 key 回退为 key 本身
    pub fn t_with_args(&self, key: &str, args: serde_json::Value) -> String  # 占位符插值（ADR-0005）
    pub fn t_with_array(&self, key: &str, args: &[&str]) -> String  # 位置参数插值 {0} {1}
}
