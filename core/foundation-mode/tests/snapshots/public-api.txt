## Traits & Implementations

impl Vertex {
    pub fn new(x: f64, y: f64) -> Self
}

impl Point2D {
    pub fn new(x: f64, y: f64) -> Self
}

impl OrientationLine {
    pub fn new(start: Point2D, end: Point2D) -> Option<Self>   // 重合两点返回 None
    pub fn length(&self) -> f64
}

impl OrientationCalculator {
    pub fn calculate(line: &OrientationLine) -> Option<Orientation>   // 产出 B1 类型
    pub fn normalize_angle(angle: f32) -> Option<Orientation>          // -90°→270°, 400°→40°
}

impl MercatorCoord {
    pub fn from_lat_lon(lat: f64, lon: f64) -> Self
    pub fn to_lat_lon(&self) -> (f64, f64)
}

impl PlaneMileUnit {
    pub fn new(x: f64, y: f64) -> Self
    pub fn distance_to(&self, other: &Self) -> f64
}

impl McBlockCoord {
    pub fn new(block_x: i32, block_z: i32) -> Self
    pub fn from_plane(plane: &PlaneMileUnit, blocks_per_meter: f64) -> Self
}

impl CoordinateConverter {
    pub fn set_center(&mut self, mercator: MercatorCoord)
    pub fn blocks_per_meter(&self) -> f64
    pub fn set_blocks_per_meter(&mut self, ratio: f64)
    pub fn mercator_to_plane(&self, mercator: MercatorCoord) -> Option<PlaneMileUnit>
    pub fn plane_to_mc(&self, plane: &PlaneMileUnit) -> McBlockCoord
    pub fn convert(&self, mercator: MercatorCoord) -> Option<(PlaneMileUnit, McBlockCoord)>
}

impl Default for CoordinateConverter

impl BoundaryDrawer {
    pub fn new() -> Self
    pub fn set_max_vertices(&mut self, count: usize)
    pub fn vertices(&self) -> &[Vertex]
    pub fn state(&self) -> BoundaryState
    pub fn handle_event(&mut self, event: BoundaryUiEvent) -> EventResult
    pub fn reset(&mut self)
    pub fn load_determined(&mut self, vertices: Vec<Vertex>)   // T24 新增：装载外部（地图确认）已验证边界
}

impl Default for BoundaryDrawer

impl std::error::Error for BoundaryValidationError (thiserror)
impl std::fmt::Display for BoundaryValidationError
