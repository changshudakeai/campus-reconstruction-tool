pub enum MaterialRole { BuildingFoundation, FoundationGround, BuildingWall, BuildingWindow, BuildingFloor, BuildingRoof, BuildingEntrance, BuildingAccent, Road, Water, TreeTrunk, TreeLeaves, SportsCourt, SportsLine, OtherRail }
pub enum MaterialError { MaterialUnavailable(String) }

impl From<MaterialError> for GenerationError

## Traits

pub trait Generator {
    type Input;
    fn category(&self) -> shared_domain_types::CandidateCategory;
    fn generate(&self, input: &Self::Input, materials: &MaterialsAdapter) -> Result<BlockModel, GenerationError>;
}
impl Generator for BuildingGenerator (Input = BuildingCandidate, category = Building)
impl Generator for RoadGenerator (Input = AreaInput, category = Road)
impl Generator for WaterGenerator (Input = AreaInput, category = Water)
impl Generator for VegetationGenerator (Input = (), category = Vegetation)
impl Generator for SportsGenerator (Input = AreaInput, category = Sports)
impl Generator for OtherGenerator (Input = OtherCandidate, category = Other)
