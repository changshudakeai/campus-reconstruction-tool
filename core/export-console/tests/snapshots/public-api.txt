    ContinueReview,
}

## Traits & Implementations

pub trait SealGate: Send + Sync {
    fn seal(&self, plan_id: &PlanId) -> Result<(), String>
    fn release(&self, plan_id: &PlanId) -> Result<(), String>
}

impl ExportRequest {
    pub fn new(plan_id: String, keep_by_category: Vec<(CandidateCategory, usize)>, keep_total: usize, pending_count: usize, remove_count: usize, keep_candidates: Vec<String>) -> Self
}

impl BoundaryExportRequest {
    pub fn new(campus_name: impl Into<String>, plan_id: shared_domain_types::PlanId, plan_name: impl Into<String>, minecraft_version: impl Into<String>, boundary: Option<shared_domain_types::Boundary>, boundary_confirmed: bool, orientation: Option<shared_domain_types::Orientation>, schematic_path: impl Into<PathBuf>, manifest_path: impl Into<PathBuf>) -> Self
    pub fn from_plan_info(plan: manifest_generator::PlanInfo, boundary: Option<shared_domain_types::Boundary>, boundary_confirmed: bool, orientation: Option<shared_domain_types::Orientation>, schematic_path: impl Into<PathBuf>, manifest_path: impl Into<PathBuf>) -> Self
}

impl ExportStage {
    pub fn label_key(self) -> &'static str
    pub fn index(self) -> u32
    pub fn from_index(index: u32) -> Self
}

impl ProgressTracker {
    pub fn new() -> Self
    pub fn set_stage(&self, stage: ExportStage)
    pub fn stage(&self) -> ExportStage
    pub fn report_percent(&self, percent: u32) -> bool
    pub fn percent(&self) -> u32
    pub fn finish(&self)
    pub fn fail(&self)
}

impl MockSealGate {
    pub fn new() -> Self
    pub fn is_sealed(&self) -> bool
}
impl SealGate for MockSealGate

impl<G: SealGate> ExportConsole<G> {
    // 缝 5：接收导出请求 + 确认弹窗
    pub fn new(gate: G) -> Self
    pub fn new_with_material_table(gate: G, material_table: manifest_generator::MaterialTable) -> Self
    pub fn export_confirmed_boundary(&mut self, request: BoundaryExportRequest) -> Result<BoundaryExportResult>
    pub fn load_request(&mut self, request: ExportRequest) -> Result<()>
    pub fn confirm_dialog_view(&self) -> Option<ExportConfirmDialogView>
    pub fn cancel(&mut self) -> Result<()>
    // 封账闸门（确认即封账；封账失败不生效）
    pub fn confirm_export(&mut self) -> Result<()>
    // 缝 6：导出执行（成功跳转 / 失败回滚 + B7 弹窗铁律）
    pub fn execute_export(&mut self, model: &BlockModel, output_path: &Path) -> Result<NavigationTarget>
    pub fn failure_target(&self) -> Option<NavigationTarget>
    // 非阻塞进度条
    pub fn progress(&self) -> &ProgressTracker
    pub fn progress_view(&self) -> ExportProgressView
    pub fn is_exporting(&self) -> bool
}

impl ExportConfirmDialogView {
    pub fn from_request(request: &ExportRequest) -> Self
}

impl ExportProgressView {
    pub fn from_tracker(tracker: &ProgressTracker) -> Self
}
